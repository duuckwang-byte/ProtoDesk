/* ═══════════════════════════════════════════════════════════════════
   灰罩诊断脚本（供用户在主窗口 DevTools Console 执行，一次定位遮罩身份）
   用法：
   1. 打开工具 → 复现问题（API 对话 → 输出完成 → 出现灰罩）
   2. 按 F12 / Ctrl+Shift+I 打开 DevTools
   3. 确认左上角上下文是「top」（主窗口，不是 iframe）
   4. 把下方 IIFE 粘贴到 Console 回车
   输出 JSON 后发回即可。
   ═══════════════════════════════════════════════════════════════════ */
(function () {
  var out = { topFixed: [], bodyClass: '', aiMask: null, aiModal: null, bigZ: [] };

  /* 1. 主窗口所有 position:fixed 且可见的元素（含 z-index/背景色） */
  document.querySelectorAll('body *').forEach(function (el) {
    if (el.children.length) return; /* 只看叶子节点避免噪音 */
    var cs = getComputedStyle(el);
    if (cs.position === 'fixed' && cs.display !== 'none') {
      out.topFixed.push({
        tag: el.tagName,
        id: el.id || '',
        cls: String(el.className || '').slice(0, 60),
        z: cs.zIndex,
        bg: cs.backgroundColor,
        w: el.offsetWidth, h: el.offsetHeight
      });
    }
  });

  /* 2. body class（排查 ctrl-inspecting 等异常态） */
  out.bodyClass = document.body.className;

  /* 3. AI 弹窗本体可见性 */
  var m = document.querySelector('.ai-modal');
  if (m) {
    var mc = getComputedStyle(m);
    out.aiModal = { display: mc.display, w: m.offsetWidth, h: m.offsetHeight };
  }

  /* 4. aiMask 状态 */
  var am = document.getElementById('aiMask');
  if (am) {
    var ac = getComputedStyle(am);
    out.aiMask = { display: ac.display, inline: am.getAttribute('style') };
  }

  /* 5. 全屏背景为暗色（灰色系）的元素 */
  out.grayFull = out.topFixed.filter(function (l) {
    return /rgba\(15, ?23, ?42|rgba\(0, ?0, ?0|rgb\(15, ?23, ?42/.test(l.bg);
  });

  return JSON.stringify(out, null, 2);
})()
