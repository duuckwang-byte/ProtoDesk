/* 增强诊断 v2：定位 4 个无 id drawer-mask 的来源（父级链 + outerHTML 片段）
   在复现出灰罩后，F12 → 上下文选 top → 粘贴执行 */
(function () {
  var out = [];
  document.querySelectorAll('.drawer-mask').forEach(function (el) {
    if (getComputedStyle(el).display === 'none') return;
    var chain = [];
    var p = el.parentElement;
    for (var i = 0; i < 4 && p; i++) {
      chain.push({ tag: p.tagName, id: p.id || '', cls: String(p.className || '').slice(0, 50) });
      p = p.parentElement;
    }
    out.push({
      id: el.id || '(无id)', cls: String(el.className || ''),
      parentChain: chain,
      html: el.outerHTML ? el.outerHTML.slice(0, 200) : ''
    });
  });
  /* 也列出所有含 drawer 的元素，确认是否有整个抽屉结构 */
  var drawers = [];
  document.querySelectorAll('[class*="drawer"]').forEach(function (el) {
    var cs = getComputedStyle(el);
    if (cs.display !== 'none' && el.offsetWidth > 0) {
      drawers.push({ id: el.id || '', cls: String(el.className || '').slice(0, 60), w: el.offsetWidth, h: el.offsetHeight, tag: el.tagName });
    }
  });
  return JSON.stringify({ masks: out, visibleDrawers: drawers }, null, 2);
})()
